cmake_minimum_required(VERSION 3.24)
project(sovereignx_vulkan_render_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/frame.comp.spv COMMAND ${Vulkan_GLSLC_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/frame.comp -o ${CMAKE_CURRENT_BINARY_DIR}/frame.comp.spv DEPENDS frame.comp)
add_custom_target(sovereignx-frame-shader DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/frame.comp.spv)
add_executable(sovereignx-vulkan-worker main.cpp ${CMAKE_CURRENT_BINARY_DIR}/frame.comp.spv)
add_dependencies(sovereignx-vulkan-worker sovereignx-frame-shader)
target_compile_definitions(sovereignx-vulkan-worker PRIVATE SOVEREIGNX_SHADER_PATH="${CMAKE_CURRENT_BINARY_DIR}/frame.comp.spv")
target_link_libraries(sovereignx-vulkan-worker PRIVATE Vulkan::Vulkan bcrypt)
add_executable(sovereignx-adl-telemetry adl-telemetry.cpp)
